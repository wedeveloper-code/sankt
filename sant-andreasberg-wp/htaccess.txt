# --- PHP settings ---
# NOTE: php_value only works with mod_php. For PHP-FPM set these in php.ini or wp-config.php:
#   define('WP_MEMORY_LIMIT', '256M');        (wp-config.php)
#   upload_max_filesize = 64M                 (php.ini / .user.ini)
#   post_max_size = 64M
#   max_execution_time = 300
<IfModule mod_php.c>

    php_value post_max_size 64M

</IfModule>

</IfModule>
